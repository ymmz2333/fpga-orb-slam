cmake_minimum_required(VERSION 3.10)
project(golden_packet_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(OpenCV 4.5.5 EXACT REQUIRED COMPONENTS core imgproc features2d)
if(NOT ORB_SOURCE)
  message(FATAL_ERROR "ORB_SOURCE required")
endif()
add_executable(golden_packet_probe main.cc "${ORB_SOURCE}/src/FeatureIO.cc" "${ORB_SOURCE}/src/ORBextractor.cc")
target_include_directories(golden_packet_probe PRIVATE "${ORB_SOURCE}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(golden_packet_probe ${OpenCV_LIBS})
target_compile_options(golden_packet_probe PRIVATE -Wall -Wextra)
