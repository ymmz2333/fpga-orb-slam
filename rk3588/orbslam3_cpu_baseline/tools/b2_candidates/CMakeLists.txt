cmake_minimum_required(VERSION 3.10)
project(orb_design_candidates LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(ORB_CANDIDATE_LAPPING_UPPER 1000 CACHE STRING "Frame ABI: 1000; archived B2 comparison only: 0")
if(NOT ORB_CANDIDATE_LAPPING_UPPER MATCHES "^(0|1000)$")
  message(FATAL_ERROR "Only corrected Frame ABI or archived B2 comparison is supported")
endif()
find_package(OpenCV 4.5.5 EXACT REQUIRED COMPONENTS core imgproc imgcodecs features2d highgui)
if(NOT ORB_SOURCE OR NOT ORB_REFERENCE)
  message(FATAL_ERROR "ORB_SOURCE and an unmodified ORB_REFERENCE snapshot are required")
endif()
add_library(original_reference OBJECT reference.cc)
target_include_directories(original_reference PRIVATE "${ORB_REFERENCE}/include" ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(original_reference PRIVATE ORB_REFERENCE_CC="${ORB_REFERENCE}/src/ORBextractor.cc" ORB_SLAM3=ORB_SLAM3_REFERENCE ORB_CANDIDATE_LAPPING_UPPER=${ORB_CANDIDATE_LAPPING_UPPER})
add_executable(orb_candidates main.cc "${ORB_SOURCE}/src/ORBextractor.cc" "${ORB_SOURCE}/src/FeatureIO.cc" $<TARGET_OBJECTS:original_reference>)
target_compile_definitions(orb_candidates PRIVATE ORB_DESIGN_CANDIDATES ORB_CANDIDATE_LAPPING_UPPER=${ORB_CANDIDATE_LAPPING_UPPER})
target_include_directories(orb_candidates PRIVATE "${ORB_SOURCE}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(orb_candidates ${OpenCV_LIBS})
target_compile_options(orb_candidates PRIVATE -Wall -Wextra)
add_executable(candidate_fixture fixture.cc "${ORB_SOURCE}/src/ORBextractor.cc")
target_compile_definitions(candidate_fixture PRIVATE ORB_DESIGN_CANDIDATES)
target_include_directories(candidate_fixture PRIVATE "${ORB_SOURCE}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(candidate_fixture ${OpenCV_LIBS})
