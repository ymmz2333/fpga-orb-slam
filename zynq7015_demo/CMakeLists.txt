cmake_minimum_required(VERSION 3.16)
project(zynq_slam LANGUAGES C CXX)
find_package(Threads REQUIRED)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ZSLAM_TURBOJPEG_INCLUDE "" CACHE PATH "Pinned libjpeg-turbo 3.2.0 src directory")
set(ZSLAM_TURBOJPEG_LIBRARY "" CACHE FILEPATH "Pinned static libturbojpeg.a")
if(NOT EXISTS "${ZSLAM_TURBOJPEG_INCLUDE}/turbojpeg.h" OR NOT EXISTS "${ZSLAM_TURBOJPEG_LIBRARY}")
  message(FATAL_ERROR "Build the pinned dependency with scripts/jpeg_dependency.py, then supply ZSLAM_TURBOJPEG_INCLUDE and ZSLAM_TURBOJPEG_LIBRARY")
endif()
add_library(zslam src/math.cpp src/pnp.cpp src/slam.cpp src/map_store.cpp src/formats.cpp src/ui.cpp src/camera.cpp src/ports.cpp)
target_include_directories(zslam PUBLIC include interfaces)
target_include_directories(zslam PUBLIC "${ZSLAM_TURBOJPEG_INCLUDE}")
target_link_libraries(zslam PUBLIC Threads::Threads "${ZSLAM_TURBOJPEG_LIBRARY}")
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(zslam PUBLIC stdc++fs)
endif()
target_compile_options(zslam PRIVATE -Wall -Wextra)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(orb_engine STATIC vendor/orb_engine/engine.c)
  target_include_directories(orb_engine PUBLIC vendor/orb_engine)
  target_link_libraries(zslam PRIVATE orb_engine)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm|armv7|armv7l)")
    target_compile_options(zslam PUBLIC -mcpu=cortex-a9 -mfpu=neon -mfloat-abi=hard)
    target_compile_options(orb_engine PRIVATE -mcpu=cortex-a9 -mfpu=neon -mfloat-abi=hard)
  endif()
endif()
add_executable(zynq-slam src/main.cpp)
target_link_libraries(zynq-slam PRIVATE zslam)
add_executable(zynq-demo src/demo_main.cpp)
target_link_libraries(zynq-demo PRIVATE zslam)
if(WIN32)
  target_link_libraries(zynq-demo PRIVATE ws2_32)
endif()
enable_testing()
add_executable(test-zslam tests/test_slam.cpp)
target_link_libraries(test-zslam PRIVATE zslam)
file(COPY tests/fixtures/camera DESTINATION "${CMAKE_BINARY_DIR}/test-fixtures")
add_test(NAME geometry_mapping_and_ports COMMAND test-zslam ${CMAKE_BINARY_DIR})
