cmake_minimum_required(VERSION 3.16)
project(orb60_live_frontend LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ORB_SOURCE "" CACHE PATH "Existing SLAM source tree")
set(ACCEPTED_LIBRARY "" CACHE FILEPATH "Preserved accepted libORB_SLAM3.so")
if(NOT EXISTS "${ACCEPTED_LIBRARY}")
  message(FATAL_ERROR "A preserved accepted SLAM library is required")
endif()
find_package(OpenCV 4.4 REQUIRED)
find_package(Eigen3 3.1 REQUIRED)
find_package(Pangolin REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
configure_file("${ACCEPTED_LIBRARY}" "${CMAKE_BINARY_DIR}/lib/libORB_SLAM3.so" COPYONLY)
file(READ "${ORB_SOURCE}/Examples/Monocular/mono_euroc.cc" LIVE_MAIN)
string(REPLACE "OrbTelemetry" "DenseFrameTelemetry" LIVE_MAIN "${LIVE_MAIN}")
file(WRITE "${CMAKE_BINARY_DIR}/live_main.cpp" "${LIVE_MAIN}")
add_executable(mono_euroc_live "${CMAKE_BINARY_DIR}/live_main.cpp" live_extension.cpp)
target_compile_definitions(mono_euroc_live PRIVATE COMPILEDWITHC11 REGISTER_TIMES)
target_compile_options(mono_euroc_live PRIVATE -O2 -march=native)
target_include_directories(mono_euroc_live PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${ORB_SOURCE}" "${ORB_SOURCE}/include"
  "${ORB_SOURCE}/include/CameraModels" "${ORB_SOURCE}/Thirdparty/Sophus" ${EIGEN3_INCLUDE_DIR} ${Pangolin_INCLUDE_DIRS})
target_link_libraries(mono_euroc_live PRIVATE "${CMAKE_BINARY_DIR}/lib/libORB_SLAM3.so" ${OpenCV_LIBS}
  ${Pangolin_LIBRARIES} "${ORB_SOURCE}/Thirdparty/DBoW2/lib/libDBoW2.so" "${ORB_SOURCE}/Thirdparty/g2o/lib/libg2o.so"
  boost_serialization crypto pthread)
set_target_properties(mono_euroc_live PROPERTIES BUILD_RPATH "${CMAKE_BINARY_DIR}/lib")
