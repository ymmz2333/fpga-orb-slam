cmake_minimum_required(VERSION 3.16)
project(orb60_dense_npu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RKNN_PRIVATE_ROOT "" CACHE PATH "Private RKNN header/runtime directory")
if(NOT EXISTS "${RKNN_PRIVATE_ROOT}/rknn_api.h" OR NOT EXISTS "${RKNN_PRIVATE_ROOT}/librknnrt.so")
    message(FATAL_ERROR "Provide matching private RKNN headers/runtime")
endif()
add_executable(dense_npu_worker worker.cpp)
target_include_directories(dense_npu_worker PRIVATE "${RKNN_PRIVATE_ROOT}")
target_link_libraries(dense_npu_worker PRIVATE "${RKNN_PRIVATE_ROOT}/librknnrt.so")
target_compile_options(dense_npu_worker PRIVATE -Wall -Wextra -Werror)
set_target_properties(dense_npu_worker PROPERTIES BUILD_RPATH "${RKNN_PRIVATE_ROOT}")
