cmake_minimum_required(VERSION 3.16)
project(kv260_ps_ops LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  message(FATAL_ERROR "Build the plugin natively on KV260")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLOG REQUIRED IMPORTED_TARGET libglog)
add_library(kv260_ps_relu SHARED relu.cpp)
target_compile_features(kv260_ps_relu PRIVATE cxx_std_17)
target_compile_options(kv260_ps_relu PRIVATE -Wall -Wextra -Werror)
target_include_directories(kv260_ps_relu SYSTEM PRIVATE "${KV_RUNTIME}/include")
target_include_directories(kv260_ps_relu PRIVATE "${KV_POLICY_DIR}")
foreach(name vitis_ai_library-cpu_task vitis_ai_library-runner_helper vart-runner xir)
  string(MAKE_C_IDENTIFIER "${name}" key)
  find_library(${key}_LIB NAMES "${name}" PATHS "${KV_RUNTIME}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(kv260_ps_relu PRIVATE "${${key}_LIB}")
endforeach()
target_link_libraries(kv260_ps_relu PRIVATE PkgConfig::GLOG)
target_link_options(kv260_ps_relu PRIVATE -Wl,--no-undefined)
set_target_properties(kv260_ps_relu PROPERTIES OUTPUT_NAME vart_op_imp_relu
  INSTALL_RPATH "${KV_RUNTIME}/lib" BUILD_WITH_INSTALL_RPATH ON)
install(TARGETS kv260_ps_relu LIBRARY DESTINATION lib)
