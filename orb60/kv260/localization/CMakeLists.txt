cmake_minimum_required(VERSION 3.16)
project(KV260_FIXED_MAP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
add_library(fixed_map_core src/math.cpp src/pnp.cpp src/slam.cpp src/map_store.cpp src/formats.cpp)
target_include_directories(fixed_map_core PUBLIC include)
target_compile_options(fixed_map_core PRIVATE -Wall -Wextra)
add_executable(fixed_map_kv260 localize_main.cc)
target_link_libraries(fixed_map_kv260 PRIVATE fixed_map_core)
if(WIN32)
    target_link_libraries(fixed_map_kv260 PRIVATE ws2_32)
endif()
